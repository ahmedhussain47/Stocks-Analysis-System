# ── Walk-forward evaluation ───────────────────────────────────────────────────

def _ets_pred(train_ret: np.ndarray, h: int = 1) -> float:
    n  = len(train_ret)
    sf = StatsForecast(
        models=[AutoETS(season_length=7)],
        freq='1h', n_jobs=1
    )
    sf_df = pd.DataFrame({
        'unique_id': ['x'] * n,
        'ds': pd.date_range('2000-01-01', periods=n, freq='1h'),
        'y': train_ret,
    })
    pred = sf.forecast(df=sf_df, h=h)
    return float(pred['AutoETS'].values[0])


def _chronos_pred(close_window: np.ndarray, h: int = 1) -> float | None:
    if _chronos is None:

    try:

        last_px = float(close_window[-1])

# ── Compute metrics & new ensemble weights ────────────────────────────────────

# Annualisation: 1H bars, step S=24 → daily rebalancing → √252
ANNUALISE = np.sqrt(252)

metrics = []
for model in ['AutoETS', 'Chronos']:
    col = f'{model}_ls_ret'
    if col not in df_wf.columns:
        continue
    ls     = df_wf[col].dropna()
    da_col = f'{model}_correct'
    da     = df_wf[da_col].mean() if da_col in df_wf.columns else float('nan')
    sharpe = float(ls.mean() / ls.std() * ANNUALISE) if ls.std() > 0 else 0.0
    metrics.append({
        'model':    model,
        'n':        len(ls),
        'dir_acc':  round(da, 4),
        'mean_ret': round(float(ls.mean()), 6),
        'sharpe':   round(sharpe, 4),
    })

metrics_df = pd.DataFrame(metrics).sort_values('sharpe', ascending=False)

# ── Rolling DirAcc over time (signal stability check) ─────────────────────────
# A model that looks good overall but has unstable DirAcc is not reliable.
ROLL = 50   # rolling window in cutoffs

correct_cols = [c for c in df_wf.columns if c.endswith('_correct')]
if correct_cols:
    fig, ax = plt.subplots(figsize=(14, 4))
    for col in correct_cols:
        model = col.replace('_correct', '')
        color = 'steelblue' if model == 'Chronos' else 'darkorange'
        roll  = df_wf[col].rolling(ROLL).mean()
        ax.plot(roll.values, lw=1.5, label=f'{model} rolling-{ROLL} DirAcc', color=color)
    ax.axhline(0.5, color='gray', ls='--', lw=1.0, label='Random (50%)')
    ax.axhline(0.52, color='green', ls=':', lw=1.0, label='Target (52%)')
    ax.set_ylim(0.3, 0.7)
    ax.set_title(f'{ASSET} {INTERVAL} — Rolling DirAcc (window={ROLL} cutoffs)')
    ax.set_xlabel('Cutoff index')
    ax.set_ylabel('Directional accuracy')
    ax.legend()
    plt.tight_layout()
    plt.show()
